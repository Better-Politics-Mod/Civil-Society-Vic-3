ciso_civsoc_process_tooling_handle_creation = {

			$ci$_creation_trigger = yes
		}

			name = ciso_civil_institutions
			target = flag:$ci$
		}
	}
	else = {
		remove_list_variable = {
			name = ciso_civil_institutions
			target = flag:$ci$
		}
	} 
}

ciso_need_process_tooling_handle_needs = {
	if = {
		limit = {
			$ne$_rfp >= $min$
		}
		add_to_variable_list = {
			name = ciso_needs
			target = flag:$ne$
		}
	}
	else = {
		remove_list_variable = {
			name = ciso_needs
			target = flag:$ne$
		}
	}
	remove_modifier = $ne$_unfulfilled
	if = {
		limit = {
			is_target_in_variable_list = {
				name = ciso_needs
				target = flag:$ne$
			}
			$ne$_rfp > $ne$_fp
		}
		set_local_variable = {
			name = unfulfilled_amount
			value = {
				value = $ne$_rfp
				subtract = $ne$_fp
				divide = $ne$_rfp
			}
		}
		add_modifier = {
			name = $ne$_unfulfilled
			multiplier = local_var:unfulfilled_amount
		}
	}
}

ciso_calculate_allocation = {
	save_scope_as = state
	
	if = {
		limit = {
			is_target_in_variable_list = {
				name = ciso_civil_institutions
				target = flag:$ci$
			}
		}
		# First we need to calculate λ
		save_scope_value_as = { 
			name = avg_sqrt_w
			value = $ci$_avg_sqrt_weight
		}

		clear_saved_scope = measure

		save_scope_value_as = {
			name = avg_shift
			value = ciso_avg_already_allocated
		}

		clear_saved_scope = measure
		# lambda = (B * avg_sqrt_w / (M / n + avg_shift)) ** 2
		save_scope_value_as = {
			name = lambda
			value = {
				value = {
					value = ciso_B
					multiply = scope:avg_sqrt_w
					divide = {
						value = {
							value = $ci$_social_impact
							divide = $ci$_num_measures
						}
						add = scope:avg_shift
					}
				}
				multiply = {
					value = ciso_B
					multiply = scope:avg_sqrt_w
					divide = {
						value = {
							value = $ci$_social_impact
							divide = $ci$_num_measures
						}
						add = scope:avg_shift
					}
				}
			}
		}
		clear_saved_scope = measure

		ciso_do_every_measure_with_ci = { ci = $ci$ } 
	}
}

ciso_apply_ms_effect = {
	if = {
		limit = {
			has_modifier = $ms$_effect
		}
		remove_modifier = $ms$_effect
	}
	if = {
		limit = {
			$ms$_investment > 0
		}
		# calculate effect value
		set_local_variable = {
			name = effect_value
			# investment * ciso_B / (ciso_B + investment)
			# this means diminishing returns on investment
			value = {
				value = $ms$_investment
				multiply = ciso_B
				divide = {
					value = ciso_B
					add = $ms$_investment
				}
			}
		}
		add_modifier = {
			name = $ms$_effect
			multiplier = local_var:effect_value
		}
	}
}

ciso_mv_atmosphere = {
	if = {
		limit = {
			NOT = { has_variable = ciso_state_atmosphere_value }
		}
		set_variable = {
			name = ciso_state_atmosphere_value
			value = 50
		}
	}
	if = {
		limit = {
			is_target_in_variable_list = {
				name = ciso_civil_institutions
				target = flag:$ci$
			}
			NOT = { $ci$_stance = 0}
		}
		change_variable = {
			name = ciso_state_atmosphere_value
			add = {
				value = $ci$_stance
				multiply = $ci$_social_impact
				round = yes
				pow = 0.5
				divide = 50
				if = {
					limit = {
						$ci$_stance > 0
					}
					multiply = -1
				}
			}
		}

		clamp_variable = {
			name = ciso_state_atmosphere_value
			min = 0
			max = 100
		}
	}

}

ciso_align_ci = {
	clear_variable_list = $ci$_aligned_groups
	if = {
		limit = {
			is_target_in_variable_list = {
				name = ciso_civil_institutions
				target = flag:$ci$
			}
		}
		ordered_in_global_list = {
			variable = ciso_alignment
			order_by = $ci$_alignment
			max = {
				value = $ci$_num_sites
			}
			prev = {
				add_to_variable_list = {
					name = $ci$_aligned_groups
					target = prev
				}
			}
		}
		set_local_variable = {
			name = counter
			value = 1
		}
		ordered_in_list = {
			variable = $ci$_aligned_groups
			clear_saved_scope = alignment
			save_scope_as = alignment
			prev = { ciso_apply_alignment = {
				ci = $ci$
			} }
			change_local_variable = {
				name = counter
				add = 1
			}
		}

	}
}
